Quickstart

g++ -std=c++17 -Wall -Wextra -Wpedantic -Iinclude \
  src/order_book.cpp src/backtest.cpp src/main.cpp \
  -o build/quant_replay.exe

build\quant_replay.exe data\sample_events.csv build\trades.csv
py python\analyze_results.py build\trades.csv
